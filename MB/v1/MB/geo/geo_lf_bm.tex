\item \fig{opggeoar_a_lf}
Vi har at
\begin{align*}
	A_{\triangle ACN}&=\frac{3\cdot2}{2} = 3 \vn
	A_{\triangle BMC}&=\frac{4\cdot2}{2}=4 \vn
	A_{\triangle ALB}&=\frac{6\cdot1}{2}=3 \vn
	A_{\square ALMN}&=3\cdot6=18
\end{align*}
Altså er
\begin{align*}
	A_{\triangle ABC}&=A_{\square ALMN}-A_{\triangle ACN}-A_{\triangle BMC}-A_{\triangle ALB}\\
	&=18-3-4-3\\ 
	&=8
\end{align*}
\item 
Firkanten er et parallellogram. \\
\textbf{Alternativ 1} \\
\fig{opggeoar_b_lf1}
Vi har at
\begin{align*}
	A_{\triangle DGP}&=\frac{3\cdot2}{2} = 3 \vn
	A_{\triangle EOF}&=\frac{3\cdot2}{2}=3 \vn
	A_{\square DOFP}&=3\cdot6=18
\end{align*}
Altså er
\begin{align}
	A_{\square DEFG}&=A_{\square DOFP}-A_{\triangle DGP}-A_{\triangle EOF} \\
	&=18-3-3 \\
	&=12
\end{align}
\textbf{Alternativ 2}
\fig{opggeoar_b_lf2}
Med $GF$ som grunnlinje, har $\triangle DFG$ høyde 3. Med $DE$ som grunnlinje har $\triangle DEF$ høyde 3. Altså er
\begin{align}
	A_{\triangle DFG}=\frac{4\cdot3}{2}=6\vn
	A_{\triangle DEF}=\frac{4\cdot3}{2}=6
\end{align}
Altså er
\begin{align}
	A_{\square DEFG}&=A_{\triangle DFG}+A_{\triangle DEF} \\
	&=6+6 \\
	&=12
\end{align}

\item Firkanten er et trapes.\\

\textbf{Alternativ 1} \\
\fig{opggeoar_c_lf1}
Vi har at
\begin{align*}
	A_{\triangle HKR}&=\frac{2\cdot2}{2}=2 \vn
	A_{\triangle IQJ}&=\frac{4\cdot2}{2}=4 \vn
	A_{\square HQJR}&=6\cdot2=12
\end{align*}
Altså er
\begin{align*}
	A_{\square HIJK}&=A_{\square HQJR}-A_{\triangle HKR}-A_{\triangle IQJ} \\
	&=12-2-4 \\
	&=6
\end{align*}
\textbf{Alternativ 2}\\

\fig{opggeoar_c_lf2}
Med $KJ$ som grunnlinje har $\triangle HJK$ høyde 2. Med $HI$ som grunnlinje har $\triangle HIJ$ høyde 2. 
Vi har at
\begin{align*}
	A_{\triangle HJK}&=\frac{4\cdot2}{2}=4 \vn
	A_{\triangle HIJ}&=\frac{2\cdot2}{2}=2
\end{align*}
Altså er 
\begin{align*}
	A_{\square HIJK}&=A_{\triangle HJK}+A_{\triangle HIJ} \\
	&=4+2 \\
	&=6
\end{align*}
