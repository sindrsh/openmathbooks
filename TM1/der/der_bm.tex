\section{Definisjoner \label{grense}}
Gitt en funksjon $ f(x) $ og to $ x $-verdier $ a $ og $ b $. Endringen til $ f $ relativ til endringen til $ x $ for disse verdiene er gitt som
\begin{equation}\label{releq}
\frac{f(b)-f(a)}{b-a}
\end{equation}
I \mb\ har vi sett at uttrykket over gir stigningstallet til linja som går gjennom punktene $ (a, f(a)) $ og $ (b, f(b)) $. I en matematisk sammenheng er det ekstra interessant å undersøke \eqref{releq} når $ b $ nærmer seg $ a $. Å \outl{derivere} innebærer å undersøke grenseverdien til denne brøken når $ b $ går mot $a$.

}
\regdef[Den deriverte\label{derdefa}]{
Gitt en funksjon $ f(x) $. Den  \outl{deriverte}\index{den deriverte}  av $ f $ i $ {x=a} $ er da gitt som
\begin{equation*}
	f'(a)=	\lim\limits_{b\to a} \frac{f(b)-f(a)}{b-a}
\end{equation*}
Alternativt kan vi skrive
\begin{equation*}
	f'(a) =\lim\limits_{h \to 0} \frac{f(a+h)-f(a)}{h}
\end{equation*}
Linja som har stigningstall $ f'(a) $, og som går gjennom punktet $ (a, f(a)) $, kalles \outl{tangeringslinja} til $ f $ for $ x=a $.
\fig{der1}
}
\eks[1]{
Gitt $ f(x)=x^2 $. Finn $ f'(2) $.

\sv
Vi har at
\algv{
	f'(2) &=\lim\limits_{h \to 0} \frac{f(2+h)-f(2)}{h} \br
	&=\lim\limits_{h \to 0} \frac{(2+h)^2-2^2}{h} \br
	&= \lim\limits_{h \to 0} \frac{2^2+4h + (h)^2-2^2}{h} \br
	&= 4
}
}
\newpage
\eks[2]{ \label{dereks2}
Gitt $ f(x)=x^3 $. Finn $ f'(a) $.

\sv
Vi har at
\alg{
f'(a) &=\lim\limits_{h \to 0} \frac{(a+h)^3-a^3}{h} \br
&= \lim\limits_{h \to 0}\frac{a^3+3a^2h +3 ah^2+ h^3-a^3}{h} \br
&= \lim\limits_{h \to 0}\left(3a^2 +3ah+h^2\right) \br
&= 3a^2
}
Altså er $ f'(a)=3a^2 $.
}

\spr{
Et annet ord for en \textit{tangeringslinje} er en \outl{tangent}\index{tangent}. \os


Hvis vi for to funksjoner $f(x)$ og $g(x)$ har at ${f'(a)=g'(a)}$ og $f(a)=g(a)$, sier vi at funksjonene tangerer hverandre for $x=a$ (eventuelt i punktet $(a, f(a))$. \os

Hvis vi for en funksjon $f(x)$ har at $f'(a)$ eksisterer, sier vi at $f$ er \outl{deriverbar}\index{deriverbarhet} for $x=a$.

}

\section{Derivasjonsregler}
\subsubsection{Den deriverte funksjonen}
\textsl{Eksempel 2} på side \pageref{dereks2} belyser noe viktig; hvis grenseverdien gitt i \dref{derdefa} eksisterer, vil $ f'(a) $ være uttrykt ved $ a $. Og selv om $ a $ betraktes som en konstant langs veien som fører til dette uttrykket, er det ingenting som hindrer oss i å etterpå behandle $ a $ som en variabel. Hvis $ f'(a) $ er et resultat av derivasjon av funksjonen $ f(x) $ er det også hendig å omdøpe $ a $ til $ x $:\regv
\reg[Den deriverte funksjonen]{
	Gitt en funksjon $ f(x) $. Den \outl{deriverte funksjonen}\index{derivert funksjon} til $ f $ er funksjonen som fremkommer ved å erstatte $ a $ i \dref{derdefa} med $ x $. Denne funksjonen skriver vi som $ f'(x) $.
}
\eks[]{
Gitt $ {f(x)=x^3} $. Siden\footnote{Se \textsl{Eksempel 2}, side \pageref{dereks2}.} $ {f'(a)=3a^2} $, er $ f'(x)=3x^2 $. 
} \regv

\spr{
Alternative skrivemåter for \sym{$f'(x)$} er \sym{$f'$}, 
\sym{$ (f)'$} og \sym{$\frac{\text{d}}{\text{d}x} f $}. \os

Den deriverte funksjonen til $f$ kalles også den \outl{førstederiverte}\index{førstederiverte} funksjonen av $f$. \vsk

En funksjon $f(x)$ som er deriverbar for alle ${x\in D_f}$ kaller vi en \outl{deriverbar funksjon}\outl{deriverbar}.
}
\info{Den deriverte av høyere orden}{
Gitt en funksjon $f$. Hvis vi deriverer $f'$, får vi den \outl{andrederiverte}\index{andrederiverte} funksjonen av $f$, som vi skriver som $f''$. Altså er ${(f')'=f''}$. På samme måte er ${(f'')'=f'''}$, ${(f''')'=f''''}$ og så videre.
}
\newpage
\reg[Deriverbarhet impliserer kontinuitet \label{derleadstocont}]{
	Gitt en funksjon $f(x)$. Hvis $f$ er deriverbar for ${x=a}$, er $f$ kontinuerlig for ${x=a}$.
}
\fork{(\ref{derleadstocont}) Deriverbarhet impliserer kontinuitet}{
	Av \dref{derdefa} har vi at
	\[ f'(a)=\lim\limits_{b\to a} \frac{f(b)-f(a)}{b-a}\]
	For at denne grensen skal eksistere, må vi ha at $ {\lim\limits_{b\to a} \left(f(b)-f(a)\right)=0} $ (hvis ikke går grensen mot $ \pm \infty $, og da er ikke den deriverte definert), og følgelig er $ {\lim\limits_{b\to a} f(b)=f(a)} $. Dermed er $ f $ kontinuerlig for $ x=a $.
}
\info{Merk}{
	Det er veldig viktig å være klar over at vilkåret i \rref{derleadstocont} ikke kan bli snudd; at en funksjon er kontinuerlig gir ingen garanti for at den er deriverbar. Men \rref{derleadstocont} gir et krav om at en funksjon må være kontinuerlig i et punkt for å være deriverbar i det samme punktet.
}
\reg[Den deriverte ved delt forskrift \label{derdeltforskrift}]{
	Gitt funksjonene $g(x)$ og $k(x)$, hvor $D_g=\mathbb{R}$ og $ D_k= \mathbb{R}$, og 
	\begin{equation*}f(x)= \left\lbrace{
			\begin{array}{rcr}
				g &,&x\leq a \br
				k &,& x> a
			\end{array}
		}\right.
	\end{equation*}
	Hvis $f$ er kontinuerlig for ${x=a}$, og ${g'(a)=k'(a)}$, er ${f'(a)=g'(a)=k'(a)}$. \vsk
	
	Hvis $g'(a)\neq k'(a)$, er ikke $f$ deriverbar for $x=a$.
}

\eks[1]{
Gitt funksjonen
	\begin{equation*}f(x)= \left\lbrace{
		\begin{array}{rcr}
			x &,&x\leq0 \br
			x^2+x &,& x> 0
		\end{array}
	}\right. 
\end{equation*}
Finn $f'(0)$ hvis verdien eksisterer.

\sv
Vi setter $g(x)=x$ og $k(x)=x^2+x$.
Vi har at
\[ \lim\limits_{x\to 0^-} g = \lim\limits_{x\to 0^+} k=0 \]
Dermed er $f$ kontinuerlig for ${x=0}$. Videre er
\[ g'(x)=1\quad,\quad k'(x)=2x+1 \]
Følgelig er $g'(0)=k'(0)=1$. Dette betyr at $f'(0)=1$.
}
\newpage
\eks[2]{
	Gitt funksjonen
	\begin{equation*}f(x)= \left\lbrace{
			\begin{array}{rcr}
				x+1 &,&x\leq0 \br
				x^2+x &,& x> 0
			\end{array}
		}\right. 
	\end{equation*}
	Finn $f'(0)$ hvis verdien eksisterer.
	
	\sv
	Vi setter $g(x)=x+1$ og $k(x)=x^2+x$.
	Vi har at
	\[ \lim\limits_{x\to 0^-} g =1 \neq \lim\limits_{x\to 0^+} k=0 \]
	Dermed er ikke $f$ kontinuerlig for ${x=0}$, og vi kan ikke bruke \rref{derdeltforskrift}, selv om $g'(0)=k'(0)$.
}

}

\info{Derivert med hensyn på}{
Derivasjon som vi har sett på så langt har vært en brøk med en differanse av $ x $-verdier i nevner og den tilknyttede differansen av $ f $-verdier i teller. Da sier vi at $ f $ er derivert med \outl{hensyn på $ x $}. I denne bokserien skal vi i all hovedsak se på funksjoner som bare er avhengige av én variabel. Gitt en funksjon $ f(x) $, er det da underforstått at \sym{$ f' $} symboliserer $ f $ derivert med hensyn på $ x $.\vsk

Samtidig er det greit å være klar over at en funksjon kan være avhengig av flere variabler. For eksempel er funksjonen
\[ f(x, y)= x^2+y^3 \]
en \outl{flervariabel funksjon}, avhengig av både $ x $ og $ y $. I dette tilfellet kan vi skrive \sym{$ \frac{\text{d}}{\text{d}x} f$} for å indikere derivasjon med hensyn på $ x $, og \sym{$ \frac{\text{d}}{\text{d}y}f $} for å indikere derivasjon med hensyn på $ y $. Da er
\[ \frac{\text{d}}{\text{d}x}f = 2x \qquad,\qquad  \frac{\text{d}}{\text{d}y}f = 3y^2 ,\]
}

\subsection*{Den deriverte av elementære funksjoner}
\reg[\derel \label{derel}]{
For en variabel $ x $ og en konstant $ r $ er
\begin{align}
(e^x)' &=e^{x} \label{derex} \vn	
(\ln |x|)' &=  \frac{1}{x} \label{derlnx} \vn
(x^r)' &= r x^{r-1} \label{derxr}
\end{align}
}	
\eks{
Deriver uttrykket $x^7$. 

\sv
\[ \left(x^7\right)'=7x^{7-1}=7x^6 \]
}
\reg[Den deriverte av sammensatte funksjoner \label{dersam}]{
Gitt en konstant $ a $ og funksjonene $ f(x) $ og $ g(x) $. Da er
\begin{align*}
	\left(a\cdot f\right)'&=a\cdot f' \vn
	\left(f+g\right)' &=f'+g' \vn
	\left(f-g\right)'&=f'-g'
\end{align*}
}
\eks{
Deriver uttrykket $2e^x-4x^2+3\ln x$.

\sv
\[ (2e^x-4x^2+3\ln x)'=2e^x-8x+\frac{3}{x} \]
}

De tre likningene følger så og si direkte av \dref{derdefa}.
}

\subsection*{Kjerne-, produkt- og divisjonsregelen ved derivasjon}
\reg[\kjernereg \label{kjerne}]{
For en funksjon $ f(x)=g\left[u(x)\right] $ har vi at
\begin{equation*}
	f'(x) = g'(u)u'(x)
\end{equation*}
}
\eks{
Finn $ f'(x) $ når  $ f(x)=e^{x^2+x+1} $.\\
\sv

Vi setter $ u=x^2+x+1 $, og får at
\[ g(u)=e^u \quad,\quad g'(u)=e^u \quad,\quad u'(x) = 2x + 1 \]
Altså er
\algv{
 f'(x) &= g'(u)u'(x) \\
 &= e^u(2x+1)\\
 &= e^{x^2+x+1}(2x+1) 
}
} \vsk

\reg[\prodregder \label{prodregder}]{
Gitt funksjonene $ f(x) $, $ u(x) $ og $ v(x) $, hvor $ {f=uv} $. Da er
\[ f'=u'v+uv' \]	
}
\eks[1]{
Finn den deriverte av funksjonen $ f(x)=x^2e^x $.

\sv
Vi setter $ {u(x)=x^2} $ og $ v(x)=e^x $. Da er
\[ f = uv\quad,\quad
u'=2x, \quad,\quad
v'=e^x \]
Altså er
\algv{
f' &= u'v+uv' \\
&=2xe^x +x^2 e^x \\
&=xe^x(2+x)
}
}
\vsk

\reg[\divder \label{divder}]{
Gitt funksjonene $ f(x) $, $ u(x) $ og $ v(x) $, hvor $ f=\dfrac{u}{v} $. Da er
\begin{equation*}\label{divdereq}
	f'=\frac{u'v-uv'}{v^2} 
\end{equation*}
}
\eks{
Finn den deriverte av funksjonen $ f(x)=\frac{\ln x}{x^4} $.

\sv
Vi setter $ u(x)=\ln x $ og $ v(x)=x^4 $. Da er
\[ f=\frac{u}{v}\quad,\quad u'=x^{-1} \quad,\quad v' = 4x^3 \]
Altså er
\algv{
f' &= \frac{u'v-uv'}{v^2}\br
&=\frac{x^{-1}\cdot x^4-\ln x\cdot 4x^3}{x^8} \br
&= \frac{1-4 \ln x}{x^5}
}
\notesm{Vi kan også finne $ f' $ ved å sette $ {u(x)=\ln x} $ og $ {v(x)=x^{-4}} $, for så å bruke produktregelen.}
}

\reg[\lhop \label{Lhop}]{
Gitt to deriverbare funksjoner $ f(x) $ og $ g(x) $, hvor
\[ f(a)=g(a)=0 \]
Da er
\nn{
\lim\limits_{x\to a} \frac{f}{g}=\lim\limits_{x\to a}\frac{f'}{g'}\label{Lhopeq}
}
}
\eks[]{
Finn grenseverdien til $ \lim\limits_{x\to 0} \dfrac{e^x-1}{x}$.

\sv
Vi setter $ {f(x)=e^x-1} $ og $ {g(x)=x} $, og merker oss at $ {f(0)=g(0)=0} $. Dermed har vi at
\alg{
\lim\limits_{x\to 0} \frac{e^x-1}{x}&=\lim\limits_{x\to0}\frac{f}{g}
\br
&=\lim\limits_{x\to0} \frac{f'}{g'}\br
&=\lim\limits_{x\to0}\frac{e^x}{1} \\
&= 1
} 
}

\section{Forklaringer}
\fork{\ref{derdeltforskrift} Den deriverte ved delt forskrift}{
Vi har at
\alg{
 g'(a)&=\lim\limits_{b\to a^-} \frac{g(b)-g(a)}{b-a}=\lim\limits_{b\to a^+} \frac{g(b)-g(a)}{b-a} \vn
 k'(a)&=\lim\limits_{b\to a^-} \frac{k(b)-k(a)}{b-a}=\lim\limits_{b\to a^+} \frac{k(b)-k(a)}{b-a} 
}
Hvis $f'(a)$ eksisterer, er
\[  f'(a)=\lim\limits_{b\to a^-} \frac{g(b)-g(a)}{b-a}=\lim\limits_{b\to a^+} \frac{k(b)-g(a)}{b-a} \]
Når $f$ er kontinuerlig, er ${\lim\limits_{b\to a} g(a)=\lim\limits_{b\to a}k(a)}$, som betyr at
\[ \lim\limits_{b\to a^+} \frac{k(b)-k(a)}{b-a}=\lim\limits_{b\to a^+} \frac{k(b)-g(a)}{b-a}\]
Siden $g'(a)=k'(a)$, er
\[ \lim\limits_{b\to a^-} \frac{g(b)-g(a)}{b-a}=\lim\limits_{b\to a^+} \frac{k(b)-k(a)}{b-a}=\lim\limits_{b\to a^+} \frac{k(b)-g(a)}{b-a} \]
Dermed er
\[ f'(a)=g'(a)=k'(a) \]
Hvis vi derimot har at $g'(a)\neq k'(a)$, er
\[ \lim\limits_{b\to a^-} \frac{g(b)-g(a)}{b-a}\neq\lim\limits_{b\to a^+} \frac{k(b)-g(a)}{b-a} \]
Dermed er ikke $f$ deriverbar for $x=a$.
}

\begin{equation}
	k(b)-k(x)=(b-x)\left[k'(x)+v_k(b) \label{kjernefork} \right]
\end{equation}
Ved å undersøke dette uttryket for tilfellet hvor ${b\to x}$, finner vi at $\lim\limits_{b\to x}v_k(b) =0$. \os

Vi setter ${t=u(x)}$ og ${s=u(b)}$. Ved å bruke uttrykket fra \eqref{kjernefork} kan vi nå skrive
\begin{align*}
	t-s&=u(b)-u(x)=(b-x)[f'(x)+v_u(b)]\vn
	f(b)-f(x)&=g(s)-g(t)=(t-s)[g'(t)+v_g(b)]
\end{align*}
Ved å sette uttrykket for $t-s$ inn i uttrykket for $f(b)-f(x)$, har vi at
\begin{align*}
	f(b)-f(x)&=(b-x)[u'(x)+v_f(b)][g'(t)+v_g(b)] \br
	\frac{f(b)-f(x)}{b-x}&=[u'(x)+v_u(b)][g'(t)+v_g(b)]
\end{align*}
Følgelig har vi at
\begin{align*}
\lim\limits_{b\to x}\frac{f(b)-f(x)}{b-x}&=\lim\limits_{b\to x} [u'(x)+v_u(b)] \lim\limits_{b\to x} [g'(t)+v_g(b)] \br
f'(x)&=u'(x)g'(u)
\end{align*}
}

\[ f'=u'v+uv' \]
}

\textbf{Likning (\ref{derlnx})}\\
Gitt at ${x>0}$, da er $ {x=e^{\ln x}} $. Vi setter $ {u=\ln x} $ og $ {g(u)=e^u} $. Da har vi at $ x=g(u) $, og at
\alg{
g'(u) &= e^u=e^{\ln x}=x \br
u'(x) &= \left(\ln x\right)'
}
Av \refunnbr{kjerne}{kjerneregelen} (\rref{kjerne}) har vi at
\alg{
\left(x\right)' &= g'(u)u'(x)\\
&= x \left(\ln x\right)'
}
Da\footnote{Se \refops{opgderxn}.}  ${\left(x\right)'=1} $, har vi at
\[ 1 = x\left(\ln x \right)' \]
Altså er
\[ \left(\ln x\right)'= \frac{1}{x} \]
for $x>0$. \os

Gitt at $x<0$, da er $x=-e^{\ln(-x)}$. Ved å sette $u=\ln(-x)$ og $g(u)=-e^u$, har vi av kjerneregelen at
\[ (x)'=-e^{\ln(-x)}\cdot (\ln (-x))' \]
Dermed er
\[ (\ln(-x))'=\frac{1}{x} \]
for $x<0$. \vsk

\textbf{Likning (\ref{derxr})}\\
Gitt at $x>0$, da har vi at
\[ (\ln x^r)'=(r \ln x)' = \frac{r}{x} \]
Vi setter $ u=x^r $. Av kjerneregelen har vi at
\alg{
	\frac{r}{x}&=	(\ln u)'\\
	&=\frac{1}{u} u' \\
	&=\frac{1}{x^r}(x^r)' 
}
Altså er
\[ 	(x^r)'= \frac{r}{x}x^r = r x^{r-1} \]
For tilfellet hvor ${x<0}$, men $x^r$ og $(-1)^r$ er definert, kan vi skrive
\[ x^r=((-1)(-x))^r=(-1)^r(-x)^r=(-1)^r\left(e^{\ln(-x)}\right)^r=(-1)^re^{r\ln (-x)} \]
Ved å sette $u=r\ln(-x)$, har vi av kjerneregelen at
\[ \left(x^r\right)' = (-1)^re^{r\ln(-x)}\frac{1}{x}=(-1)^r(-x)^r\frac{r}{x}=rx^{r-1} \]
}
