\regdef[Halveringslinje]{
Gitt $ \angle BAC $. For et punkt $ P $ på \outl{halveringslinja}\index{halveringslinje} til vinkelen (blå linje i figuren) er
\begin{equation*}
	\angle BAP = \angle PAC=\frac{1}{2}\angle BAC
\end{equation*}
\fig{halfline}
}	
\regdef[Midtpunkt]{
\outl{Midtpunktet}\index{midtpunkt} $ C $ til $ AB $ er punktet på linjestykket slik at $ {AC=CB} $.
\fig{midtp} 
}
\regdef[Midtnormal]{
\outl{Midtnormalen}\index{midtnormal} til $ AB $ (blå linje i figuren) står normalt på $AB$, og går gjennom midtpunktet til $ AB $.
\fig{midnorm0}
}

\regdef[sin, cos og tan I \label{sincostan}]{\index{sin} \index{cos} \index{tan}
Gitt en rettvinklet trekant med katetene $ a $ og $ b $, hypotenus $ c $, og vinkel $ v $, som vist i figuren under.
\fig{sincostan}
Da er \vs
\begin{align*}
\sin v &= \frac{a}{c} \vn
\cos v &= \frac{b}{c} \vn
\tan v &= \frac{a}{b} 	
\end{align*}
}
\spr{
I figuren over blir $ a $ kalt den \outl{motstående} kateten til vinkel $ v $, og $ b $ den \outl{hosliggende} kateten til vinkel $v$.\vsk

sin, cos og tan er forkortelser for henholdsvis \outl{sinus}\index{sinus}, \outl{cosinus}\index{cosinus} og \outl{tangens}\index{tangens}.
}

\info{Eksaktverdier}{
	De aller fleste sinus-, cosinus- og tangensverdier er irrasjonale tall, i praktiske anvendelser av verdiene er det derfor vanlig å benytte digitale hjelpemidler. Verdiene som er viktigst for teoretiske formål er gitt i \refved{sincoseksakt}.
}

\[ \sin v =\frac{3}{5}\quad,\quad \cos v = \frac{4}{5}\quad,\quad \tan v = \frac{3}{4} \]
}\vsk

\reg[sin, cos og tan II \label{sincostan2}]{ 
	Gitt $ \triangle ABC$, hvor $ {v=\angle BAC>90^\circ} $, som vist i figuren under.
	\fig{sincostan2}
	Da er \vs
\begin{align*}
	\sin v &= \frac{CD}{AC}  \vn
	\cos v &= -\frac{AD}{AC} \vn
	\tan v &= -\frac{CD}{AD}
\end{align*}
}
\eks[]{ \vsb
\fig{sincostaneks2}
I figuren over er $ CD=\sqrt{3} $, $ AD=1 $ og $ AC=2 $. Da er
\[ \sin 120^\circ =\frac{\sqrt{3}}{2}\quad,\quad \cos 120^\circ = -\frac{1}{2}\quad,\quad \tan 120^\circ =-\sqrt{3} \]
}

\reg[\tang \label{tang}]{
	En linje som skjærer en sirkel i bare ett punkt, kalles en \\\outl{tangent}\index{tangent} til sirkelen.\vsk
	
	La $ S $ være sentrum i en sirkel, og la $ A $ være skjæringspunktet til denne sirkelen og en linje. Da har vi at
	\nn{
		\text{linja er en tangent til sirkelen} \Longleftrightarrow AS\text{ står vinkelrett på linja} 
	}
	
	\fig{tang0a}
}

	En \outl{periferivinkel}\index{periferivinkel} har toppunkt på sirkelbuen.\vsk
	
	Gitt en periferivinkel $ u $ og en sentralvinkel $ v $, som er innskrevet i samme sirkel og som spenner over samme sirkelbue. Da er\vs

		ABG + AFG &= ABG + BGE \\
		AFG &= BGE

	\alg{
		AFG+GFC+GEC&= AEC \\
		GEC &= \frac{1}{6}ABC
	}

	\[HEC=\frac{1}{6}ABC \]

\fork{\ref{innskrsirk} \innskrsirk}{
	\fig{innskr1b}
	\textbf{Det kan innskrives en sirkel som tangerer sidene} \os
	Gitt $ \triangle ABC $. Vi lar $ S $ være skjæringspunktet til de respektive halveringslinjene til $ \angle BAC $ og $ \angle CBA $. Videre plasserer vi $ D $, $ E $ og $ F $ slik at $ {DS\perp AB} $, $ {ES\perp BC} $ og $ {FS\perp AC} $. $ {\triangle ASD\cong\triangle ASF} $ fordi begge er rettvinklede og har hypotenus $ AS $, og $ {\angle DAS=\angle SAF} $. Tilsvarende er $ {\triangle BSD \cong \triangle BSE} $. Dermed er $ {SE=SD=SF} $. Sirkelen med sentrum i $S$ og radius $ SE $ vil dermed tangere sidene til $\triangle ABC$ i $ F $, $ D $ og $ E $. \vsk
	
	\textbf{Halveringslinjene møtes i sentrum i den innskrevne sirkelen} \os
	Vi har at $ {\triangle CSE \cong  \triangle CSF} $, fordi begge er rettvinklede og har hypotenus $ CS $, og $ {SF=SE} $. Altså er $ {\angle FCS=\angle ECS}$, som betyr at $ CS $ ligger på halveringslinja til $ \angle ACB $.
}

\fork{\ref{tang} \tang}{
\textbf{\boldmath Linja er en tangent til sirkelen ${\Rightarrow AS}$ står vinkelrett på linja}
\fig{tangforkla}
Vi antar at vinkelen mellom linja og $ AS $ er ulik $ 90^\circ $. Da må det finnes et punkt $ B $ på linja slik at $ {\angle BAS=\angle SBA} $, som betyr at $ \triangle ASB $ er likebeint. Følgelig er $ {AS=BS} $, og da $ AS $ er lik radien i sirkelen, må dette bety at $ B $ også ligger på sirkelen. Dette motsier at $ A $ er det eneste skjæringspunktet til sirkelen og linja, og dermed må vinkelen mellom linja og $ AS $ være $ 90^\circ $. \vsk

\textbf{\boldmath $AS$ står vinkelrett på linja $\Rightarrow$ Linja er en tangent til sirkelen}
\fig{tangforklb}
Gitt et vilkårlig punkt $ B $, som ikke samsvarer med $ A $, på linja. Da er $ BS $ hypotenusen i $ \triangle ABC $. Dette innebærer at $ BS $ er større enn radien til sirkelen ($ {BS>AS} $), og da kan $ B $ umulig ligge på sirkelen. Altså er $ A $ det eneste punktet som ligger på både linja og sirkelen, og dermed er linja en tangent til sirkelen.
}

\fork{\ref{perfvink} \perfvink}{
	Tilhørende periferi- og sentralvinkler kan deles inn i tre tilfeller.\vsk
	
	\textbf{(i) En diameter i sirkelen er høyre eller venstre vinkelbein i begge vinklene}\\[5pt]
	I figuren under er $ S $ sentrum i sirkelen, ${\angle BAC= u}$ en periferi-vinkel og $ {\angle BSC=v} $ den tilhørende sentralvinkelen. Vi setter $ {\angle {SCB}=a} $. $ {\angle {ACS}=\angle {SAC}=u} $ og ${\angle {CBS}=\angle {SCB}=a} $ fordi både $ \triangle {ASC} $ og $ \triangle {SBC} $ er likebeinte.\vspace{-1pt}
	\fig{pervinkforkla}
	Vi har at \vs
	\begin{align}
		2a &= 180^\circ-v \label{aogv} \vn
		2u+2a &= 180^\circ \label{uoga}
	\end{align}
	Vi setter uttrykket for $ 2a $ fra \eqref{aogv} inn i \eqref{uoga}:
	\alg{
		2u+180^\circ -v &= 180^\circ\\
		2u &= v
	}
	
	\textbf{(ii) Vinklene ligger innenfor samme halvdel av sirkelen}\\[5pt]
	I figuren under er $ u $ en periferivinkel og $ v $ den tilhørende sentralvinkelen. I tillegg har vi tegnet inn en diameter, som er med på å danne vinklene $ a $ og $ b $. Både $ u $ og $ v $ ligger i sin helhet på samme side av denne diameteren.
	\fig{pervinkforkld}
	Ettersom $  {u+a }$  er en periferivinkel, og $ {v+b }$ den tilhørende sentralvinkelen, vet vi av tilfelle 1 at
	\[ 2(u+a) = v+b  \]
	Men ettersom $ a $ og $ b $ også er samhørende periferi- og sentralvinkler, er $ {2a=b} $. Det betyr at
	\algv{
		2u+b &= v+b \\
		2u &= v
	}
	\newpage
	\textbf{(iii) Vinklene ligger ikke innenfor samme halvdel av sirkelen}\\[5pt]
	I figuren under er $ u $ en periferivinkel og $ v $ den tilhørende sentralvinkelen. I figuren til høyre har vi tegnet inn en diameter. Den deler $ u $ inn i vinklene $ a $ og $ c $, og $ v $ inn i $ b $ og $ d $.
	\begin{center}
		\includegraphics[]{\figp{pervinkforklc}}\qquad
		\includegraphics[]{\figp{pervinkforklb}}
	\end{center}
	$ a $ og $ c $ er begge periferivinkler, med henholdsvis $ b $ og $ d $ som tilhørende sentralvinkler. Av tilfelle i) har vi da at
	\algv{
		2a &= b \vn
		2c &= d
	}
	Dermed er
	\algv{
		2a+2c &= b+d \\
		2(a+c) &= v \\
		2u &= v
	}
}
