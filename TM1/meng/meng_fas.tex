\section*{Kapittel \ref{Meng}}

\op{opgmengterm}
Erstatt \sym{?} med \sym{$\iff$}, \sym{$\Rightarrow$} eller \sym{$\Leftarrow$}.
\abc{
	\item $a$ og $b$ er partall $\Rightarrow$ $a+b$ er et partall.
	\item tallet er delelig med 2 $\Leftarrow$ tallet er delelig med 4.
	\item tallet er delelig med både 2 og 3 $\iff$ tallet er delelig med 6.
}
