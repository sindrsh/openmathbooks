\op{opgmengterm}
Erstatt \sym{?} med \sym{$\iff$}, \sym{$\Rightarrow$} eller \sym{$\Leftarrow$}.
\abc{
\item $a$ og $b$ er partall $\sym{?}$ $a+b$ er et partall.
\item tallet er delelig med 2 $\sym{?}$ tallet er delelig med 4.
\item tallet er delelig med både 2 og 3 $\sym{?}$ tallet er delelig med 6.
}

} \os

\item $ f(x)= \frac{2}{x-3} $
