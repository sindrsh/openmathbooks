\abc{
	\item $ 1+2+3+\ldots+n = \dfrac{n(n+1)}{2} $

	\item $ 1^2 + 2^2+3^2+ \ldots+ n^2 = \dfrac{n(2n+1)(n+1)}{6} $ 
}
