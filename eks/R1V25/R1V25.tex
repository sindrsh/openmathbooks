\subsection*{Oppgave 2}
\abc{
\item Da $e^x\neq 0$ for alle $x$, er $g(x)=0$ når $2x-1=0$, altså når $x=\frac{1}{2}$.
\item Vi setter $u=e^x$ og $v=(2x-1)^2=4x^2-4x+1$. Da er $u'=e^x$ og $v'=8x-4=4(2x-1)$. Av produktregelen ved derivasjon har vi at
\begin{align}
	g'(x)=\left(\frac{1}{2}uv\right)'&=\frac{1}{2}\left(u'v+uv'\right)\\
	&=\frac{1}{2}e^x(v+v')\\
	&=\frac{1}{2}e^x\left(4x^2-4x+1+4(2x-1)\right) \\
	&=\frac{1}{2}e^x\left(4x^2-4x-3\right)
\end{align}
Ved å gange ut parantesene kan vi bekrefte at $(2x+1)(2x-3)=4x^2-4x-3$, og dermed har vi vist det vi skulle.
\item Vi lager et fortegnsskjema for $g'(x)$. Da $\frac{1}{2}e^x>0$, er det bare de to andre faktorene som bidrar til at fortegnet forandrer seg.
\begin{figure}
	\centering
	\includegraphics{opg2del1}
\end{figure}
At $g'(x)$ skifter fra positivt til negativt fortegn i $x=-\frac{3}{2}$, betyr at dette er et maksimalpunkt for $g$. Tilsvarende er $x=\frac{1}{2}$ et minimumspunkt for $g$. Videre er 
\begin{align*}
 g\left(-\frac{3}{2}\right) &= \frac{1}{2}e^{-\frac{3}{2}}\left(2\cdot\left(-\frac{3}{2}\right)-1\right)^2=2e^{-\frac{3}{2}}\vn
 g\left(\frac{1}{2}\right)&=\frac{1}{2}e^{\frac{1}{2}}\left(2\cdot\frac{1}{2}-1\right)^2=0
\end{align*}
$g$ har altså toppunkt $\left(-\frac{3}{2}, 2e^{-\frac{3}{2}}\right)$ og bunnpunkt $\left(\frac{1}{2}, 0\right)$
}

\subsection*{Oppgave 3}
\abc{
\item 
\begin{align*}
	3^{3x+2}-5&=76 \\
	3^{3x}\cdot3^2&=81 \\
	3^{3x}&=\frac{3^4}{3^{2}} \\
	3^{3x}&=3^{2}
\end{align*}
Altså er $3x=2$, og da er $x=\frac{2}{3}$.
\item 
\begin{align*}
	3\lg x+2\lg x^2 + \lg \frac{1}{x^9}&=2 \\
	3\lg x + 2\cdot2\lg x -9\lg x&= 2 \\
	-2\lg x &=2 \\
	\lg x &= -1 
\end{align*}
Antatt at $\lg$ har 10 som base, får vi at $x=\frac{1}{10}$.
}

\subsection*{Oppgave 5}
\abc{
\item Vi setter $g(x)=x^2+2$ og $h(x)=2e^x$. Vi har at 
\begin{align*}
	\lim\limits_{x\to0^-} g&= \lim\limits_{x\to0^-} 0^2+2=2 \br
	\lim\limits_{x\to0^+} h&=\lim\limits_{x\to0^+} 2e^0=2
\end{align*}
Da grenseverdiene er like, er $f$ kontinuerlig i $x=0$.
\item Vi har at $g'(x)=2x$ og $h'(x)=2e^x$. Da er $g'(0)=0$ og $h'(0)=2$. Dermed er ikke funksjonen deriverbar.
}

\subsection*{Oppgave 6}
\abc{
\item Vi har at $\vv{AN}=[-1-1, 2-1]=[-2, 1]$. Dermed er $|\vv{AN}|=\sqrt{(-2)^2+1^2}=\sqrt{5}$. Avstanden mellom Nils og Ahmad er $\sqrt{5}\enh{m}$.
\item Vi setter $B=(-1, a)$, da er $\vv{JB}=[-1, a]$. Kravet om parallellitet tilsier at $\vv{JB}$ er et multiplum av $\vv{AN}$. $2\vv{JB}=\vv{AN}$ hvis $a=\frac{1}{2}$.
\item Vi setter $M=(x, y)$. Da er $\vv{JM}=[x, y]$ og $\vv{AM}=[x-1, y-1]$. 
Av ortogonaliteten har vi at
\begin{align*}
	\vv{AM}\cdot\vv{JM}&=0\\
	[x-1, y-1]\cdot[1, 1]&=0 \\
	x-1+y-1&=0 \\
	y &= 2-x 
\end{align*}
Av lengden har vi at
\begin{align*}
x^2+y^2&=10 \\
x^2+(2-x)^2&=10 \\
x^2+4-4x+x^2 &= 10 \\
x^2-2x+3 &= 0
\end{align*}
Da $3=1\cdot(-3)$ og $1+(-3)=-2$, er 
\[ (x+1)(x-3)=0 \]
Dermed er $M=(-1, 3)$ eller $M=(3, -1)$. Av de to ligger $N$ nærmest $(-1, 3)$, og derfor er $M=(-1, 3)$.
}

\subsection*{Oppgave 4}
\abc{
\item I CAS definerer vi $\vec{r}$ (celle 1). Farten er lengden til $r'(t)$ (celle 2). Omgjort til knop er farten ca. 5,1 (celle 3).
\item Vi finner funksjonen som gir avstanden mellom $\vec{r}$ og $(4, 7)$ (celle 4). I celle 5 f finner vi bunnpunktet til funksjonen, som gir at den minste avstanden mellom fiskebåten og fyret er ca. 0,95\enh{km}.
\item Posisjonen $\vec{p}(t)$ til fiskestimen er gitt ved $\vec{p}(t)=[1+4t, -3+11t]$. Av celle 6 finner vi at $\vec{r}$ og $\vec{p}$ bare har lik $x$-verdi når $t=0$. Men når $t=0$ er $\vec{p}=[1, -3]$, mens $\vec{r}=[1, 4]$, og da er $y$-verdiene forskjellige. Fiskebåten vil ikke treffe fiskestimen.
\item  Posisjonen $\vec{m}(t)$ til den andre fiskebåten er gitt ved \\ $\vec{m}(t)=[-2+6at, 4at]$. Skal denne fiskebåten treffe fiskestimen, må de to likningene i celle 7 og 8 være oppfylt. Ved å løse dette likningssystemet (celle 9), finner vi at $a=\frac{3}{2}$. Da er
\[ \vec{m}'(t)=\frac{3}{2}[6, 4]=3[3, 2] \]
Videre er
\[ |3[3,2]|=3\sqrt{3^2+2^2}=3\sqrt{13}\approx10,81 \]
}
Altså må denne båten ha $10,81\enh{km/h}$ som fart for å treffe fiskestimen.

{\footnotesize \textit{Alternativ metode for oppgave b}\\
Avstanden $h$ mellom et punkt $B$ og en linje gitt av punktene $A$ og retningsvektoren $\vec{v}$ er gitt som
\[ h=\frac{|\det\left(\vv{AB}, \vec{v}\right)|}{|\vec{v}|} \]
Linjen $\vec{r}$ er gitt av punktet $A=(1, 4)$ og retningsvektoren $\vec{v}=\vec{r}\,'(t)=[5, 8]$. Vi setter $B=(4, 7)$, da er $\vv{AB}=[3, 3]$. Ved å skrive \texttt{h=Determinant\{\{3,3\},\{5, 8\}\}/|(5, 8)|} i CAS, får vi at
\[ h=\frac{9}{\sqrt{89}}\approx 0,95 \]
Dermed er den minste avstanden mellom fiskebåten og fyret ca. 0,95\enh{km}.
}
